p_patient_id    IN NUMBER,
p_department_id IN NUMBER,
p_admission_id  OUT NUMBER) IS
v_bed_id beds.bed_id%TYPE;
BEGIN
SELECT bed_id
INTO v_bed_id
FROM beds
WHERE department_id = p_department_id
AND status = 'AVAILABLE'
FETCH FIRST 1 ROW ONLY
FOR UPDATE SKIP LOCKED;
UPDATE beds
SET status = 'OCCUPIED'
WHERE bed_id = v_bed_id;
INSERT INTO admissions(patient_id, bed_id, admitted_at, status)
VALUES (p_patient_id, v_bed_id, SYSTIMESTAMP, 'ACTIVE')
RETURNING admission_id INTO p_admission_id;
EXCEPTION
WHEN NO_DATA_FOUND THEN
RAISE_APPLICATION_ERROR(-20001, 'NO_BED_AVAILABLE');
WHEN OTHERS THEN
RAISE;
END;
